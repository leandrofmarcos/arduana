import React, { useState, useEffect } from 'react';
import { Ship, FileCheck, ClipboardList, Search, Scale, DollarSign, CheckCircle, Truck, Package } from 'lucide-react';

const CustomsJourney = () => {
  const [currentStep, setCurrentStep] = useState(0);
  const [animateToken, setAnimateToken] = useState(false);

  const steps = [
    {
      id: 0,
      title: "Início",
      subtitle: "Preparação",
      description: "Documentação reunida e licenças obtidas",
      icon: FileCheck,
      color: "from-blue-400 to-blue-600",
      position: { x: 50, y: 550 }
    },
    {
      id: 1,
      title: "Porto",
      subtitle: "Chegada",
      description: "Mercadoria chegou ao território nacional",
      icon: Ship,
      color: "from-cyan-400 to-cyan-600",
      position: { x: 200, y: 550 }
    },
    {
      id: 2,
      title: "Registro DI",
      subtitle: "Declaração",
      description: "Declaração de Importação registrada",
      icon: ClipboardList,
      color: "from-purple-400 to-purple-600",
      position: { x: 350, y: 550 }
    },
    {
      id: 3,
      title: "Parametrização",
      subtitle: "Sistema",
      description: "Canal de conferência definido",
      icon: Search,
      color: "from-indigo-400 to-indigo-600",
      position: { x: 500, y: 550 }
    },
    {
      id: 4,
      title: "Fiscalização",
      subtitle: "Conferência",
      description: "Análise documental e física",
      icon: Scale,
      color: "from-orange-400 to-orange-600",
      position: { x: 650, y: 550 }
    },
    {
      id: 5,
      title: "Tributos",
      subtitle: "Pagamento",
      description: "Impostos e taxas quitados",
      icon: DollarSign,
      color: "from-green-400 to-green-600",
      position: { x: 800, y: 400 }
    },
    {
      id: 6,
      title: "Desembaraço",
      subtitle: "Liberação",
      description: "Mercadoria nacionalizada",
      icon: CheckCircle,
      color: "from-teal-400 to-teal-600",
      position: { x: 800, y: 250 }
    },
    {
      id: 7,
      title: "Entrega",
      subtitle: "Finalização",
      description: "Mercadoria liberada para retirada",
      icon: Truck,
      color: "from-emerald-400 to-emerald-600",
      position: { x: 650, y: 100 }
    }
  ];

  useEffect(() => {
    setAnimateToken(true);
    const timer = setTimeout(() => setAnimateToken(false), 600);
    return () => clearTimeout(timer);
  }, [currentStep]);

  const nextStep = () => {
    if (currentStep < steps.length - 1) {
      setCurrentStep(currentStep + 1);
    }
  };

  const prevStep = () => {
    if (currentStep > 0) {
      setCurrentStep(currentStep - 1);
    }
  };

  const resetJourney = () => {
    setCurrentStep(0);
  };

  // Generate path between steps
  const generatePath = () => {
    let path = `M ${steps[0].position.x + 60} ${steps[0].position.y + 60}`;
    for (let i = 1; i < steps.length; i++) {
      const prev = steps[i - 1].position;
      const curr = steps[i].position;
      const midX = (prev.x + curr.x) / 2;
      const midY = (prev.y + curr.y) / 2;
      path += ` Q ${midX} ${midY}, ${curr.x + 60} ${curr.y + 60}`;
    }
    return path;
  };

  const current = steps[currentStep];
  const IconComponent = current.icon;

  return (
    <div className="w-full min-h-screen bg-gradient-to-br from-indigo-900 via-purple-900 to-pink-900 overflow-x-hidden">
      
      {/* Decorative elements */}
      <div className="absolute inset-0 opacity-10 pointer-events-none">
        <div className="absolute top-20 left-20 w-64 h-64 bg-white rounded-full blur-3xl"></div>
        <div className="absolute bottom-20 right-20 w-96 h-96 bg-blue-300 rounded-full blur-3xl"></div>
      </div>

      {/* Header */}
      <div className="relative z-10 p-4 md:p-8">
        <div className="max-w-7xl mx-auto">
          <h1 className="text-3xl md:text-5xl font-bold text-white mb-2 flex items-center gap-3">
            <Package className="w-8 h-8 md:w-12 md:h-12" />
            Jornada Aduaneira
          </h1>
          <p className="text-purple-200 text-sm md:text-lg">DI 24/1234567-8 • Importador: Empresa XYZ Ltda</p>
        </div>
      </div>

      {/* Game Board - Scrollable */}
      <div className="relative z-10 overflow-x-auto pb-32" style={{ minHeight: '700px' }}>
        <div className="relative mx-auto" style={{ width: '900px', height: '700px' }}>
          <svg width="900" height="700" className="absolute">
            {/* Path line */}
            <path
              d={generatePath()}
              fill="none"
              stroke="rgba(255, 255, 255, 0.2)"
              strokeWidth="8"
              strokeDasharray="20,10"
            />
            
            {/* Progress line */}
            <path
              d={generatePath()}
              fill="none"
              stroke="url(#gradient)"
              strokeWidth="8"
              strokeDasharray="20,10"
              strokeDashoffset="0"
              style={{
                strokeDasharray: '2000',
                strokeDashoffset: 2000 - (2000 * (currentStep / (steps.length - 1)))
              }}
              className="transition-all duration-1000 ease-in-out"
            />
            
            <defs>
              <linearGradient id="gradient" x1="0%" y1="0%" x2="100%" y2="0%">
                <stop offset="0%" stopColor="#10b981" />
                <stop offset="100%" stopColor="#3b82f6" />
              </linearGradient>
            </defs>
          </svg>

          {/* Steps */}
          {steps.map((step, index) => {
            const StepIcon = step.icon;
            const isCompleted = index < currentStep;
            const isCurrent = index === currentStep;
            const isPending = index > currentStep;

            return (
              <div
                key={step.id}
                className="absolute transition-all duration-500"
                style={{
                  left: `${step.position.x}px`,
                  top: `${step.position.y}px`,
                  transform: isCurrent && animateToken ? 'scale(1.2)' : 'scale(1)'
                }}
              >
                {/* Step circle */}
                <div className={`relative w-32 h-32 rounded-full shadow-2xl flex items-center justify-center
                  ${isCurrent ? 'ring-8 ring-yellow-400 ring-opacity-50 animate-pulse' : ''}
                  ${isCompleted ? 'bg-gradient-to-br from-green-400 to-green-600' : ''}
                  ${isCurrent ? `bg-gradient-to-br ${step.color}` : ''}
                  ${isPending ? 'bg-gradient-to-br from-gray-600 to-gray-800' : ''}
                  transition-all duration-500`}
                >
                  <StepIcon className="w-16 h-16 text-white" />
                  
                  {isCompleted && (
                    <div className="absolute -top-2 -right-2 bg-green-500 rounded-full p-2 shadow-lg">
                      <CheckCircle className="w-6 h-6 text-white" />
                    </div>
                  )}
                </div>

                {/* Step label */}
                <div className={`absolute -bottom-20 left-1/2 transform -translate-x-1/2 text-center w-40
                  ${isCurrent ? 'scale-110' : 'scale-100'} transition-all duration-300`}>
                  <p className="font-bold text-white text-lg">{step.title}</p>
                  <p className="text-purple-200 text-sm">{step.subtitle}</p>
                </div>

                {/* Current step indicator */}
                {isCurrent && (
                  <div className="absolute -top-24 left-1/2 transform -translate-x-1/2 bg-yellow-400 text-gray-900 px-4 py-2 rounded-full font-bold shadow-lg animate-bounce whitespace-nowrap">
                    VOCÊ ESTÁ AQUI
                  </div>
                )}
              </div>
            );
          })}
        </div>
      </div>

      {/* Bottom Panel - Fixed */}
      <div className="fixed bottom-0 left-0 right-0 z-20 bg-gradient-to-r from-gray-900 to-gray-800 border-t-4 border-purple-500 shadow-2xl">
        <div className="max-w-7xl mx-auto p-4 md:p-6">
          <div className="flex flex-col lg:flex-row items-start lg:items-center justify-between gap-4">
            {/* Current step info */}
            <div className="flex-1 w-full lg:w-auto">
              <div className="flex items-center gap-3 md:gap-4 mb-3">
                <div className={`w-12 h-12 md:w-16 md:h-16 rounded-xl bg-gradient-to-br ${current.color} flex items-center justify-center shadow-lg flex-shrink-0`}>
                  <IconComponent className="w-6 h-6 md:w-8 md:h-8 text-white" />
                </div>
                <div>
                  <h2 className="text-xl md:text-2xl font-bold text-white">{current.title}</h2>
                  <p className="text-purple-300 text-sm md:text-base">{current.subtitle}</p>
                </div>
              </div>
              <p className="text-gray-300 text-sm md:text-lg mb-3">{current.description}</p>
              <div className="flex flex-wrap items-center gap-2">
                <div className="bg-purple-600 text-white px-3 py-1 rounded-full text-xs md:text-sm font-semibold">
                  Etapa {currentStep + 1} de {steps.length}
                </div>
                <div className="bg-blue-600 text-white px-3 py-1 rounded-full text-xs md:text-sm font-semibold">
                  {Math.round(((currentStep + 1) / steps.length) * 100)}% Completo
                </div>
              </div>
            </div>

            {/* Controls */}
            <div className="flex gap-2 md:gap-3 w-full lg:w-auto">
              <button
                onClick={prevStep}
                disabled={currentStep === 0}
                className="flex-1 lg:flex-none bg-gray-700 hover:bg-gray-600 disabled:bg-gray-800 disabled:opacity-50 text-white px-4 md:px-6 py-2 md:py-3 rounded-xl text-sm md:text-base font-semibold transition-all duration-200 shadow-lg disabled:cursor-not-allowed"
              >
                ← Anterior
              </button>
              <button
                onClick={nextStep}
                disabled={currentStep === steps.length - 1}
                className="flex-1 lg:flex-none bg-gradient-to-r from-purple-600 to-pink-600 hover:from-purple-500 hover:to-pink-500 disabled:from-gray-700 disabled:to-gray-800 disabled:opacity-50 text-white px-4 md:px-6 py-2 md:py-3 rounded-xl text-sm md:text-base font-semibold transition-all duration-200 shadow-lg disabled:cursor-not-allowed"
              >
                Próximo →
              </button>
              <button
                onClick={resetJourney}
                className="bg-orange-600 hover:bg-orange-500 text-white px-4 md:px-6 py-2 md:py-3 rounded-xl text-sm md:text-base font-semibold transition-all duration-200 shadow-lg"
              >
                🔄
              </button>
            </div>
          </div>
        </div>
      </div>
    </div>
  );
};

export default CustomsJourney;
